Accurate Webhook Verification Guide

Every webhook includes the X-CashApp-Signature header: t=timestamp,v1=sha256_hmac. Verify it to ensure payments cannot be spoofed! 

<?php
// verify-cashapp-webhook.php
$webhookSecret = 'YOUR_WEBHOOK_SECRET'; // whsec_...

$payload = file_get_contents('php://input');
$sigHeader = $_SERVER['HTTP_X_CASHAPP_SIGNATURE'] ?? '';

// Parse header: t=1793218000,v1=9f83ab...
parse_str(str_replace(',', '&', $sigHeader), $sigParts);
$timestamp = $sigParts['t'] ?? '';
$signature = $sigParts['v1'] ?? '';

// Recompute HMAC-SHA256
$expected = hash_hmac('sha256', "{$timestamp}.{$payload}", $webhookSecret);

if (hash_equals($expected, $signature)) {
    $event = json_decode($payload, true);
    if ($event['event'] === 'payment.completed') {
        $paymentId = $event['data']['payment_id'];
        $amountUsd = $event['data']['amount_usd'];
        // Mark order as PAID in your database!
    }
    http_response_code(200);
    echo json_encode(['received' => true]);
} else {
    http_response_code(400);
    echo 'Invalid signature verification';
}
?>
const crypto = require('crypto');

app.post('/api/cashapp-webhook', (req, res) => {
  const secret = process.env.CASHAPP_WEBHOOK_SECRET;
  const sigHeader = req.headers['x-cashapp-signature'];
  const rawBody = JSON.stringify(req.body);

  const parts = Object.fromEntries(sigHeader.split(',').map(s => s.split('=')));
  const timestamp = parts.t;
  const signature = parts.v1;

  const expected = crypto.createHmac('sha256', secret)
    .update(`${timestamp}.${rawBody}`)
    .digest('hex');

  if (signature === expected) {
    const { event, data } = req.body;
    if (event === 'payment.completed') {
      console.log(`Payment confirmed: $${data.amount_usd} (${data.amount_sats} Sats)`);
      // Update your database order status to completed
    }
    return res.status(200).json({ received: true });
  }

  return res.status(400).send('Invalid signature');
});
import hmac, hashlib
from flask import Flask, request, jsonify

app = Flask(__name__)
WEBHOOK_SECRET = "YOUR_WEBHOOK_SECRET"

@app.route('/api/cashapp-webhook', methods=['POST'])
def cashapp_webhook():
    sig_header = request.headers.get('X-CashApp-Signature', '')
    parts = dict(x.split('=') for x in sig_header.split(','))
    timestamp, signature = parts.get('t'), parts.get('v1')
    
    signed_payload = f"{timestamp}.{request.get_data(as_text=True)}".encode('utf-8')
    expected = hmac.new(WEBHOOK_SECRET.encode(), signed_payload, hashlib.sha256).hexdigest()
    
    if hmac.compare_digest(signature, expected):
        data = request.json
        if data.get('event') == 'payment.completed':
            print("Verified Cash App payment:", data['data']['amount_usd'])
        return jsonify(received=True), 200
        
    return "Invalid signature", 400
