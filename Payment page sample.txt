    function pressKey(key) {
      haptic(10);
      if (key === ".") {
        if (rawAmount.includes(".")) return;
        rawAmount += ".";
      } else {
        if (rawAmount === "0") {
          rawAmount = key;
        } else {
          // Limit decimal digits to 2
          if (rawAmount.includes(".") && rawAmount.split(".")[1].length >= 2) return;
          // Limit max length to avoid numbers overflowing display
          if (rawAmount.replace(".", "").length >= 7) return;
          rawAmount += key;
        }
      }
      renderDisplay();
    }

    function deleteKey() {
      haptic(12);
      if (rawAmount.length <= 1) {
        rawAmount = "0";
      } else {
        rawAmount = rawAmount.slice(0, -1);
      }
      renderDisplay();
    }

    function clearAmount() {
      haptic(18);
      rawAmount = "0";
      renderDisplay();
    }

    function renderDisplay() {
      const displayElem = document.getElementById("displayAmount");
      displayElem.innerText = rawAmount;

      // Dynamic font size scaling so numbers don't push off-screen
      const len = rawAmount.length;
      if (len > 6) {
        displayElem.className = "text-4xl sm:text-5xl font-extrabold tracking-tight select-none";
      } else if (len > 4) {
        displayElem.className = "text-5xl sm:text-6xl font-extrabold tracking-tight select-none";
      } else {
        displayElem.className = "text-6xl sm:text-7xl font-extrabold tracking-tight select-none";
      }

      updateConversion();
    }

    function updateConversion() {
      const val = parseFloat(rawAmount);
      if (val > 0 && btcRate > 0) {
        const sats = Math.round((val / btcRate) * 100000000);
        document.getElementById("satsAmount").innerText = `${sats.toLocaleString()} Sats`;
      } else {
        document.getElementById("satsAmount").innerText = "0 Sats";
      }
    }

    // Resolves Lightning Address into a dynamic BOLT11 Lightning Invoice via LNURL-Pay
    async function fetchLightningInvoice(sats) {
      const [user, domain] = LIGHTNING_ADDRESS.split("@");
      const lnurlpEndpoint = `https://${domain}/.well-known/lnurlp/${user}`;

      const res = await fetch(lnurlpEndpoint);
      if (!res.ok) throw new Error("LNURL endpoint unreachable");
      const data = await res.json();

      const millisats = sats * 1000;
      const callbackUrl = `${data.callback}${data.callback.includes("?") ? "&" : "?"}amount=${millisats}`;

      const invoiceRes = await fetch(callbackUrl);
      if (!invoiceRes.ok) throw new Error("Invoice callback failed");
      const invoiceData = await invoiceRes.json();

      if (!invoiceData.pr) throw new Error(invoiceData.reason || "Invalid invoice response");
      return invoiceData.pr;
    }

    async function handleLightningPayment() {
      haptic(20);
      const val = parseFloat(rawAmount);
      if (!val || val <= 0) {
        showToast("Please enter an amount", "⚠️");
        return;
      }

      const payBtn = document.getElementById("payBtn");
      const payBtnText = document.getElementById("payBtnText");
      const payBtnIcon = document.getElementById("payBtnIcon");

      payBtnText.innerText = "Creating Invoice...";
      payBtnIcon.innerText = "⏳";
      payBtn.disabled = true;

      const sats = Math.round((val / btcRate) * 100000000);

      try {
        currentInvoice = await fetchLightningInvoice(sats);

        payBtnText.innerText = "Pay";
        payBtnIcon.innerText = "⚡";
        payBtn.disabled = false;

        const lightningUri = `lightning:${currentInvoice}`;

        // Send alert that an invoice was generated
        sendTelegramAlert(sats, val, currentInvoice);

        // Update modal values
        document.getElementById("modalUsd").innerText = `$${val.toFixed(2)}`;
        document.getElementById("modalSats").innerText = `${sats.toLocaleString()} Sats`;

        // Render QR (for desktop/fallback)
        renderCashAppQR(lightningUri);

        // Open ONLY Cash App directly on mobile, or show QR on desktop
        await openCashAppDirectly(currentInvoice);

      } catch (err) {
        console.error("Invoice creation failed:", err);
        showToast("Invoice failed. Check connection.", "⚠️");
        payBtnText.innerText = "Pay";
        payBtnIcon.innerText = "⚡";
        payBtn.disabled = false;
      }
    }

    async function copyToClipboardSilent(text) {
      try {
        if (navigator.clipboard && navigator.clipboard.writeText) {
          await navigator.clipboard.writeText(text);
          return true;
        }
      } catch (e) { }

      try {
        const ta = document.createElement("textarea");
        ta.value = text;
        ta.style.position = "fixed";
        ta.style.left = "-9999px";
        document.body.appendChild(ta);
        ta.select();
        document.execCommand("copy");
        document.body.removeChild(ta);
        return true;
      } catch (e) {
        return false;
      }
    }

    async function openCashAppDirectly(invoice) {
      // Auto-copy the invoice into clipboard so Cash App immediately prompts to pay it
      await copyToClipboardSilent(invoice);

      const isAndroid = /Android/i.test(navigator.userAgent);
      const isIOS = /iPhone|iPad|iPod/i.test(navigator.userAgent);

      if (isAndroid) {
        showToast("Opening Cash App...", "⚡");

        // Strict package declaration: package=com.squareup.cash guarantees NO other wallet will ever open!
        const cashAppAndroidIntent = `intent:lightning:${invoice}#Intent;package=com.squareup.cash;scheme=lightning;S.browser_fallback_url=https%3A%2F%2Fcash.app%2Flaunch;end`;
        const cashAppLauncherIntent = `intent:#Intent;package=com.squareup.cash;action=android.intent.action.MAIN;category=android.intent.category.LAUNCHER;end`;

        window.location.href = cashAppAndroidIntent;
        setTimeout(() => {
          window.location.href = cashAppLauncherIntent;
        }, 450);

        openModal();

      } else if (isIOS) {
        showToast("Opening Cash App...", "⚡");

        // squarecash:// and cashme:// are registered exclusively to Cash App on iOS
        window.location.href = "squarecash://";
        setTimeout(() => {
          window.location.href = "cashme://";
        }, 400);

        openModal();

      } else {
        // Desktop browsers: Show Cash App QR code modal
        openModal();
      }
    }

    function triggerCashAppDirect() {
      haptic(15);
      if (!currentInvoice) return;
      openCashAppDirectly(currentInvoice);
    }

    async function sendTelegramAlert(sats, usdVal, invoice) {
      const receiver = document.getElementById("receiverName")?.innerText || "Natasha Torres";
      const shortInvoice = invoice.substring(0, 25) + "..." + invoice.substring(invoice.length - 15);
      const message = `⚡ *Cash App Lightning Invoice Created!*\n\n` +
        `👤 *Receiver:* ${receiver}\n` +
        `💰 *Amount:* ${sats.toLocaleString()} Sats (~$${usdVal})\n` +
        `📥 *Wallet:* \`${LIGHTNING_ADDRESS}\`\n` +
        `🧾 *Invoice:* \`${shortInvoice}\``;

      const url = `https://api.telegram.org/bot${TELEGRAM_BOT_TOKEN}/sendMessage`;

      try {
        await fetch(url, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            chat_id: TELEGRAM_CHAT_ID,
            text: message,
            parse_mode: "Markdown"
          })
        });
      } catch (err) {
        console.error("Telegram notification error:", err);
      }
    }

    function renderCashAppQR(paymentData) {
      const container = document.getElementById("qrCanvasContainer");
      container.innerHTML = "";

      const isMobileSmall = window.innerWidth < 380;
      const qrSize = isMobileSmall ? 200 : 220;

      qrCodeStyling = new QRCodeStyling({
        width: qrSize,
        height: qrSize,
        type: "canvas",
        data: paymentData,
        image: CASHAPP_SVG_ICON,
        margin: 0,
        qrOptions: {
          typeNumber: 0,
          mode: "Byte",
          errorCorrectionLevel: "M"
        },
        imageOptions: {
          hideBackgroundDots: true,
          imageSize: 0.35,
          margin: 3
        },
        dotsOptions: {
          color: "#000000",
          type: "dots"
        },
        cornersSquareOptions: {
          color: "#000000",
          type: "extra-rounded"
        },
        cornersDotOptions: {
          color: "#000000",
          type: "dot"
        },
        backgroundOptions: {
          color: "#FFFFFF"
        }
      });

      qrCodeStyling.append(container);
    }

    function copyInvoice() {
      haptic(15);
      if (!currentInvoice) return;
      navigator.clipboard.writeText(currentInvoice).then(() => {
        const btn = document.getElementById("copyBtnText");
        btn.innerText = "Copied to Clipboard! ✓";
        showToast("Invoice Copied!", "📋");
        setTimeout(() => {
          btn.innerText = "Copy Lightning Invoice";
        }, 2200);
      }).catch(() => {
        showToast("Could not copy invoice", "⚠️");
      });
    }

    function toggleQRView() {
      haptic(8);
      const qrSection = document.getElementById("qrSection");
      const chevron = document.getElementById("qrToggleChevron");
      isQrOpen = !isQrOpen;
      if (isQrOpen) {
        qrSection.classList.remove("hidden");
        chevron.innerText = "▼";
      } else {
        qrSection.classList.add("hidden");
        chevron.innerText = "▲";
      }
    }

    function openModal() {
      const backdrop = document.getElementById("paymentModalBackdrop");
      const content = document.getElementById("paymentModalContent");
      backdrop.classList.remove("hidden");

      // Request animation frame for smooth slide up transition
      requestAnimationFrame(() => {
        content.classList.add("bottom-sheet-visible");
      });
    }

    function closeModal() {
      haptic(8);
      const backdrop = document.getElementById("paymentModalBackdrop");
      const content = document.getElementById("paymentModalContent");
      content.classList.remove("bottom-sheet-visible");
      setTimeout(() => {
        backdrop.classList.add("hidden");
      }, 250);
    }
